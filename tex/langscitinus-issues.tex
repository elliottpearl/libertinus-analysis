% --- Dispatcher: maps word → Unicode combining mark ---
\newcommand{\markdispatch}[1]{%
  \IfStrEq{#1}{acute}{\char"0301}{}%
  \IfStrEq{#1}{grave}{\char"0300}{}%
  \IfStrEq{#1}{caron}{\char"030C}{}%
  \IfStrEq{#1}{doubleacute}{\char"030B}{}%
  \IfStrEq{#1}{doublegrave}{\char"030F}{}%
}

% --- Builder: vowel + CGJ + tilde-below + mark ---
\newcommand{\voweltildemark}[2]{%
  #1\char"034F\char"0330%
  \markdispatch{#2}%
}

  \item  "i" not losing its dot when there is a tone diacritic AND a tilde below

  \item troubles with capital I

\Itildegrave\

\textbf{\textit{á}}
\textbf{\textit{a̋}}

		á\
